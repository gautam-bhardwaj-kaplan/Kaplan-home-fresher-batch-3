🎯 Key Implementation Notes
Streak Calculation Logic:
javascript// Pseudo-code for streak calculation
function updateStreak(userId, submissionDate) {
  const lastAttempt = user.lastAttemptDate;
  const daysDiff = Math.floor((submissionDate - lastAttempt) / (1000 * 60 * 60 * 24));
  
  if (daysDiff === 1) {
    // Consecutive day
    user.currentStreak += 1;
    user.longestStreak = Math.max(user.currentStreak, user.longestStreak);
  } else if (daysDiff > 1) {
    // Streak broken
    user.currentStreak = 1;
  }
  // daysDiff === 0 means same day (no change)
  
  user.lastAttemptDate = submissionDate;
  checkBadgeEligibility(user);
}
